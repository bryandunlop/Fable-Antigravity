```ts
export interface DomainGroup { domain: Domain; label: string; primary: NavEntry[]; more: NavEntry[] }
export const DOMAIN_ORDER: Domain[] = ['home', 'flight-ops', 'scheduling', 'inflight', 'inventory', 'maintenance', 'safety', 'documents', 'admin'];
export function domainsForRole(userRole: string, additionalRoles: string[] = []): DomainGroup[] {
  const visible = entriesForRoles(userRole, additionalRoles).filter((e) => e.sidebar !== false);
  return DOMAIN_ORDER.map((domain) => {
    const in_ = visible.filter((e) => e.domain === domain);
    return { domain, label: DOMAIN_LABELS[domain], primary: in_.filter((e) => e.primary), more: in_.filter((e) => !e.primary) };
  }).filter((d) => d.primary.length + d.more.length > 0);
}
```
